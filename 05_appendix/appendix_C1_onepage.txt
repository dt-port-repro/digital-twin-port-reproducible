┌─────────────────────────────────────────────────────────────────┐
│              附录C.1  数据适配器配置模板（一页纸版）               │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  adapter: {name: adapter, version: 1.0}                          │
│  system: {type: TOS, vendor: <厂商>, product: <系统名>}          │
│                                                                  │
├─ 连接配置 ──────────────────────────────────────────────────────┤
│  conn: {protocol: REST_API, url: <URL>,                           │
│         auth: OAuth2/{id: <ID>, secret: ${SECRET}},              │
│         timeout: 30s, retry: 3次(×1.5)}                          │
│  sync: {interval: 60s, batch: 1000, initial: true}               │
│                                                                  │
├─ 字段映射 ──────────────────────────────────────────────────────┤
│  实体         源表              字段映射                         │
│  container    <表名>    {id, weight, size, type}→源字段           │
│                        转换: weight×0.4536 | size enum[20,40,45] │
│  vessel       <表名>    {id, eta, etd, berth}→源字段             │
│                        时间格式: yyyy-MM-dd HH:mm:ss             │
│  yard         <表名>    {block, bay, row, tier, cid, status}     │
│                        状态码: A→OCCUPIED / R→RESERVED / E→FREE  │
│                                                                  │
├─ 校验规则 ──────────────────────────────────────────────────────┤
│  required: [container_id, size, weight]                          │
│  rules: range(weight, 0, 30000) | not_null(id) | enum(size)     │
│                                                                  │
├─ 异常处理 ──────────────────────────────────────────────────────┤
│  failure→retry(3次)→log | skip limit: 10                        │
│  alert: [连接失败, 数据损坏, 模式不匹配]                         │
│  notify: email + webhook                                         │
│                                                                  │
├─ 性能优化 ──────────────────────────────────────────────────────┤
│  prefetch: on(100) | cache: on(300s, 10K) | compress: on        │
│                                                                  │
├─────────────────────────────────────────────────────────────────┤
│  说明：用<尖括号>替换为实际TOS系统参数即可复用，框架结构通用。      │
└─────────────────────────────────────────────────────────────────┘
